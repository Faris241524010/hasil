Contoh query untuk mendapatkan lima query dengan total waktu eksekusi tertinggi:

SELECT
    query,
    calls,
    ROUND(total_exec_time::numeric, 2) AS total_ms,
    ROUND(mean_exec_time::numeric, 2) AS mean_ms
FROM pg_stat_statements
ORDER BY total_exec_time DESC
LIMIT 5;

Untuk memeriksa indeks, jalankan:
SELECT
    tablename,
    indexname,
    indexdef
FROM pg_indexes
WHERE schemaname = 'public'
  AND tablename IN (
      'product',
      'orders',
      'customer',
      'invoice'
  )
ORDER BY tablename, indexname;
 tablename |      indexname      |                                  indexdef

 invoice   | idx_invoice_seller  | CREATE INDEX idx_invoice_seller ON public.invoice USING btree (seller_id)
